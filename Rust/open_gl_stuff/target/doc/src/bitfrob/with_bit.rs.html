1macro_rules! impl_with_bit {
2  ($fn_name:ident, $t:ty) => {
3    /// Replaces the `b` bit in `u`.
4    ///
5    /// ## Panics
6    /// * `b` can't exceed the number of bits in the type.
7    #[inline]
8    #[must_use]
9    #[cfg_attr(feature = "track_caller", track_caller)]
10    pub const fn $fn_name(b: u32, u: $t, is_set: bool) -> $t {
11      assert!(b < <$t>::BITS);
12      let mask = 1 << b;
13      (u & !mask) | ((is_set as $t) << b)
14    }
15  };
16}
17

22impl_with_bit!(u128_with_bit, u128);

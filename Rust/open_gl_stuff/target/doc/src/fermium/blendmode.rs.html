1//! Support module for pixel blending.
2
3/// A newtype'd blend operation value.
4///
5/// See `SDL_BLENDMODE_*` for examples.
6#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
7#[repr(transparent)]
8pub struct SDL_BlendMode(pub u32);
9
10/// no blending
11/// ```txt
12/// dstRGBA = srcRGBA
13/// ```
14pub const SDL_BLENDMODE_NONE: SDL_BlendMode = SDL_BlendMode(0x00000000);
15/// alpha blending
16/// ```txt
17/// dstRGB = (srcRGB * srcA) + (dstRGB * (1-srcA))
18/// dstA = srcA + (dstA * (1-srcA))
19/// ```
20pub const SDL_BLENDMODE_BLEND: SDL_BlendMode = SDL_BlendMode(0x00000001);
21/// additive blending
22/// ```txt
23/// dstRGB = (srcRGB * srcA) + dstRGB
24/// dstA = dstA
25/// ```
26pub const SDL_BLENDMODE_ADD: SDL_BlendMode = SDL_BlendMode(0x00000002);
27/// color modulate
28/// ```txt
29/// dstRGB = srcRGB * dstRGB
30/// dstA = dstA
31/// ```
32pub const SDL_BLENDMODE_MOD: SDL_BlendMode = SDL_BlendMode(0x00000004);
33/// color multiply
34/// ```txt
35/// dstRGB = (srcRGB * dstRGB) + (dstRGB * (1-srcA))
36/// dstA = (srcA * dstA) + (dstA * (1-srcA))
37/// ```
38pub const SDL_BLENDMODE_MUL: SDL_BlendMode = SDL_BlendMode(0x00000008);
39/// Sentinel for an invalid blend mode.
40pub const SDL_BLENDMODE_INVALID: SDL_BlendMode = SDL_BlendMode(0x7FFFFFFF);
41
42/// A newtype'd blend operation value.
43///
44/// See `SDL_BLENDOPERATION_*` for examples.
45#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
46#[repr(transparent)]
47pub struct SDL_BlendOperation(pub u32);
48
49/// Op: dst + src: supported by all renderers
50pub const SDL_BLENDOPERATION_ADD: SDL_BlendOperation = SDL_BlendOperation(0x1);
51/// Op: dst - src : supported by D3D9, D3D11, OpenGL, OpenGLES
52pub const SDL_BLENDOPERATION_SUBTRACT: SDL_BlendOperation =
53  SDL_BlendOperation(0x2);
54/// Op: src - dst : supported by D3D9, D3D11, OpenGL, OpenGLES
55pub const SDL_BLENDOPERATION_REV_SUBTRACT: SDL_BlendOperation =
56  SDL_BlendOperation(0x3);
57/// Op: min(dst, src) : supported by D3D11
58pub const SDL_BLENDOPERATION_MINIMUM: SDL_BlendOperation =
59  SDL_BlendOperation(0x4);
60/// Op: max(dst, src) : supported by D3D11
61pub const SDL_BLENDOPERATION_MAXIMUM: SDL_BlendOperation =
62  SDL_BlendOperation(0x5);
63
64/// A newtype'd blend factor value.
65///
66/// See the `SDL_BLENDFACTOR_*` constants for values of this type.
67#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
68#[repr(transparent)]
69pub struct SDL_BlendFactor(pub u32);
70
71/// RGBA: 0, 0, 0, 0
72pub const SDL_BLENDFACTOR_ZERO: SDL_BlendFactor = SDL_BlendFactor(0x1);
73/// RGBA: 1, 1, 1, 1
74pub const SDL_BLENDFACTOR_ONE: SDL_BlendFactor = SDL_BlendFactor(0x2);
75/// RGBA: srcR, srcG, srcB, srcA
76pub const SDL_BLENDFACTOR_SRC_COLOR: SDL_BlendFactor = SDL_BlendFactor(0x3);
77/// RGBA: 1-srcR, 1-srcG, 1-srcB, 1-srcA
78pub const SDL_BLENDFACTOR_ONE_MINUS_SRC_COLOR: SDL_BlendFactor =
79  SDL_BlendFactor(0x4);
80/// RGBA: srcA, srcA, srcA, srcA
81pub const SDL_BLENDFACTOR_SRC_ALPHA: SDL_BlendFactor = SDL_BlendFactor(0x5);
82/// RGBA: 1-srcA, 1-srcA, 1-srcA, 1-srcA
83pub const SDL_BLENDFACTOR_ONE_MINUS_SRC_ALPHA: SDL_BlendFactor =
84  SDL_BlendFactor(0x6);
85/// RGBA: dstR, dstG, dstB, dstA
86pub const SDL_BLENDFACTOR_DST_COLOR: SDL_BlendFactor = SDL_BlendFactor(0x7);
87/// RGBA: 1-dstR, 1-dstG, 1-dstB, 1-dstA
88pub const SDL_BLENDFACTOR_ONE_MINUS_DST_COLOR: SDL_BlendFactor =
89  SDL_BlendFactor(0x8);
90/// RGBA: dstA, dstA, dstA, dstA
91pub const SDL_BLENDFACTOR_DST_ALPHA: SDL_BlendFactor = SDL_BlendFactor(0x9);
92/// RGBA: 1-dstA, 1-dstA, 1-dstA, 1-dstA
93pub const SDL_BLENDFACTOR_ONE_MINUS_DST_ALPHA: SDL_BlendFactor =
94  SDL_BlendFactor(0xA);

96extern "C" {
97  /// Create a custom blend mode, which may or may not be supported by a given
98  /// renderer
99  ///
100  /// * `srcColorFactor`: source color factor
101  /// * `dstColorFactor`: destination color factor
102  /// * `colorOperation`: color operation
103  /// * `srcAlphaFactor`: source alpha factor
104  /// * `dstAlphaFactor`: destination alpha factor
105  /// * `alphaOperation`: alpha operation
106  ///
107  /// The result of the blend mode operation will be:
108  /// ```txt
109  /// dstRGB = dstRGB * dstColorFactor colorOperation srcRGB * srcColorFactor
110  /// ```
111  /// and
112  /// ```txt
113  /// dstA = dstA * dstAlphaFactor alphaOperation srcA * srcAlphaFactor
114  /// ```
115  pub fn SDL_ComposeCustomBlendMode(
116    srcColorFactor: SDL_BlendFactor, dstColorFactor: SDL_BlendFactor,
117    colorOperation: SDL_BlendOperation, srcAlphaFactor: SDL_BlendFactor,
118    dstAlphaFactor: SDL_BlendFactor, alphaOperation: SDL_BlendOperation,
119  ) -> SDL_BlendMode;
120}

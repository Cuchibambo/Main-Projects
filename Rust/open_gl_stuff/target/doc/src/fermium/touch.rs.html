1//! Module for touch interface interactions.
2
3use crate::{c_float, c_int, stdinc::*};
4
5/// Used with touch events.
6#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
7#[repr(transparent)]
8pub struct SDL_TouchID(pub Sint64);
9
10/// Used to identify a finger object.
11#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
12#[repr(transparent)]
13pub struct SDL_FingerID(pub Sint64);
14
15/// The types of a touch device.
16///
17/// See `SDL_TOUCH_DEVICE_*` constants.
18#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
19#[repr(transparent)]
20pub struct SDL_TouchDeviceType(pub i32);
21
22/// The "invalid" touch type. Used for error values and such.
23pub const SDL_TOUCH_DEVICE_INVALID: SDL_TouchDeviceType =
24  SDL_TouchDeviceType(-1);
25/// Touch screen with window-relative coordinates
26pub const SDL_TOUCH_DEVICE_DIRECT: SDL_TouchDeviceType = SDL_TouchDeviceType(0);
27/// Trackpad with absolute device coordinates
28pub const SDL_TOUCH_DEVICE_INDIRECT_ABSOLUTE: SDL_TouchDeviceType =
29  SDL_TouchDeviceType(1);
30/// Trackpad with screen cursor-relative coordinates
31pub const SDL_TOUCH_DEVICE_INDIRECT_RELATIVE: SDL_TouchDeviceType =
32  SDL_TouchDeviceType(2);
33
34/// Info for a finger object.
35#[derive(Debug, Clone, Copy, Default, PartialEq, PartialOrd)]
36#[repr(C)]
37#[allow(missing_docs)]
38pub struct SDL_Finger {
39  pub id: SDL_FingerID,
40  pub x: c_float,
41  pub y: c_float,
42  pub pressure: c_float,
43}
44
45/// Used as the device ID for *mouse events* simulated with touch input.
46pub const SDL_TOUCH_MOUSEID: u32 = -1_i32 as u32;
47
48/// Used as the `SDL_TouchID` for *touch events* simulated with mouse input.
49pub const SDL_MOUSE_TOUCHID: SDL_TouchID = SDL_TouchID(-1);
50
51extern "C" {
52  /// Get the number of registered touch devices.
53  pub fn SDL_GetNumTouchDevices();
54
55  /// Get the touch ID with the given index, or 0 if the index is invalid.
56  pub fn SDL_GetTouchDevice(index: c_int) -> SDL_TouchID;
57
58  /// Get the type of the given touch device.
59  pub fn SDL_GetTouchDeviceType(touchID: SDL_TouchID) -> SDL_TouchDeviceType;
60
61  /// Get the number of active fingers for a given touch device.
62  pub fn SDL_GetNumTouchFingers(touchID: SDL_TouchID) -> c_int;
63
64  /// Get the finger object of the given touch, with the given index.
65  pub fn SDL_GetTouchFinger(
66    touchID: SDL_TouchID, index: c_int,
67  ) -> *mut SDL_Finger;
68}

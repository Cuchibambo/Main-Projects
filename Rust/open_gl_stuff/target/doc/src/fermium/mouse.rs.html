60  /// Retrieve the current state of the mouse.
61  ///
62  /// The current button state is returned as a button bitmask, which can be
63  /// tested using the SDL_BUTTON(X) macros. `x` and `y` are set to the mouse
64  /// cursor position relative to the focus window for the currently selected
65  /// mouse.
66  ///
67  /// You can safely pass NULL for either `x` or `y`.
68  pub fn SDL_GetMouseState(x: *mut c_int, y: *mut c_int) -> u32;
69
70  /// Get the current state of the mouse, in relation to the desktop.
71  ///
72  /// This works just like [`SDL_GetMouseState`], but the coordinates will be
73  /// reported relative to the top-left of the desktop. This can be useful if
74  /// you need to track the mouse outside of a specific window and
75  /// [`SDL_CaptureMouse`] doesn't fit your needs. For example, it could be
76  /// useful if you need to track the mouse while dragging a window, where
77  /// coordinates relative to a window might not be in sync at all times.
78  ///
79  /// **Note:** [`SDL_GetMouseState`] returns the mouse position as SDL
80  /// understands it from the last pump of the event queue. This function,
81  /// however, queries the OS for the current mouse position, and as such, might
82  /// be a slightly less efficient function. Unless you know what you're doing
83  /// and have a good reason to use this function, you probably want
84  /// [`SDL_GetMouseState`] instead.
85  ///
86  /// * `x` Returns the current X coord, relative to the desktop. Can be NULL.
87  /// * `y` Returns the current Y coord, relative to the desktop. Can be NULL.
88  ///
89  /// **Return:** The current button state as a bitmask, which can be tested
90  /// using the SDL_BUTTON(X) macros.
91  pub fn SDL_GetGlobalMouseState(x: *mut c_int, y: *mut c_int) -> u32;
92
93  /// Retrieve the relative state of the mouse.
94  ///
95  /// The current button state is returned as a button bitmask, which can be
96  /// tested using the SDL_BUTTON(X) macros, and x and y are set to the mouse
97  /// deltas since the last call to `SDL_GetRelativeMouseState`.
98  pub fn SDL_GetRelativeMouseState(x: *mut c_int, y: *mut c_int) -> u32;

121  /// Set relative mouse mode.
122  ///
123  /// * `enabled` Whether or not to enable relative mode
124  ///
125  /// **Returns:** 0 on success, or -1 if relative mode is not supported.
126  ///
127  /// While the mouse is in relative mode, the cursor is hidden, and the driver
128  /// will try to report continuous motion in the current window. Only relative
129  /// motion events will be delivered, the mouse position will not change.
130  ///
131  /// In other words, this is what you'd use for an "FPS" style interface.
132  ///
133  /// **Note:** This function will flush any pending mouse motion.
134  ///
135  /// See Also: [`SDL_GetRelativeMouseMode`]
136  pub fn SDL_SetRelativeMouseMode(enabled: SDL_bool) -> c_int;
137
138  /// Capture the mouse, to track input outside an SDL window.
139  ///
140  /// * `enabled` Whether or not to enable capturing
141  ///
142  /// Capturing enables your app to obtain mouse events globally, instead of
143  /// just within your window. Not all video targets support this function. When
144  /// capturing is enabled, the current window will get all mouse events, but
145  /// unlike relative mode, no change is made to the cursor and it is not
146  /// restrained to your window.
147  ///
148  /// This function may also deny mouse input to other windows--both those in
149  /// your application and others on the system--so you should use this function
150  /// sparingly, and in small bursts. For example, you might want to track the
151  /// mouse while the user is dragging something, until the user releases a
152  /// mouse button. It is not recommended that you capture the mouse for long
153  /// periods of time, such as the entire time your app is running.
154  ///
155  /// While captured, mouse events still report coordinates relative to the
156  /// current (foreground) window, but those coordinates may be outside the
157  /// bounds of the window (including negative values). Capturing is only
158  /// allowed for the foreground window. If the window loses focus while
159  /// capturing, the capture will be disabled automatically.
160  ///
161  /// While capturing is enabled, the current window will have the
162  /// `SDL_WINDOW_MOUSE_CAPTURE` flag set.
163  ///
164  /// **Return:** 0 on success, or -1 if not supported.
165  pub fn SDL_CaptureMouse(enabled: SDL_bool) -> c_int;
166

172  /// Create a cursor, using the specified bitmap data and mask (in MSB format).
173  ///
174  /// The cursor width must be a multiple of 8 bits.
175  ///
176  /// The cursor is created in black and white according to the following:
177  ///
178  /// | data | mask | resulting pixel on screen |
179  /// |:-:|:-:|:-|
180  /// |  0   |  1   | White |
181  /// |  1   |  1   | Black |
182  /// |  0   |  0   | Transparent |
183  /// |  1   |  0   | Inverted color if possible, black if not. |
184  ///
185  /// See Also: [`SDL_FreeCursor`]
186  pub fn SDL_CreateCursor(
187    data: *const Uint8, mask: *const Uint8, w: c_int, h: c_int, hot_x: c_int,

227/// Used as a mask when testing buttons in `buttonstate`.
228///
229/// * Button 1:  Left mouse button
230/// * Button 2:  Middle mouse button
231/// * Button 3:  Right mouse button
232#[inline]
233#[must_use]
234pub const fn SDL_BUTTON(x: u32) -> u32 {
235  1 << (x - 1)
236}
237

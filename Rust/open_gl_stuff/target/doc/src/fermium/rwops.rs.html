1//! Allows for interaction with things you can read and write from.
2//!
3//! Usually files, but you can also point this at static memory and stuff in
4//! some situations.
5//!
6//! Only a minimal amount of this module is currently bound. If you want more
7//! added, PRs accepted.
8
9use crate::{c_char, c_void};
10
11/// SDL's read/write abstraction.
12///
13/// This is necessary for interaction with some of the other SDL systems, but
14/// I'm dubious as to its general value compared to using `std::fs` most of the
15/// time. So, currently, there's only bindings to allocate one from a file path
16/// and to free it later.
17#[repr(transparent)]
18pub struct SDL_RWops(c_void);
19
20extern "C" {
21  /// Creates a [`SDL_RWops`] from a filename and mode string.
22  ///
23  /// * `file` supports unicode filenames if it's a utf8-encoded name given,
24  ///   even if the local OS doesn't use utf8 encoding for filenames (it's
25  ///   recoded for you).
26  /// * The `mode` string works like with unix `fopen`, basically.
27  /// * Both strings are null-terminated, of course.
28  pub fn SDL_RWFromFile(
29    file: *const c_char, mode: *const c_char,
30  ) -> *mut SDL_RWops;
31
32  /// Free an [`SDL_RWops`] after use.
33  ///
34  /// Some functions already have a flag that they will free an `SDL_RWops`
35  /// given after they use it, so this isn't always necessary.
36  pub fn SDL_FreeRW(area: *mut SDL_RWops);
37}

7/// SDL uses this for key events.
8///
9/// Key events are for raw key events. For translated textual input you want the
10/// "text input" events instead.
11#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
12#[repr(C)]
13pub struct SDL_Keysym {
14  /// SDL physical key code
15  pub scancode: SDL_Scancode,
16  /// SDL virtual key code
17  pub sym: SDL_Keycode,
18  /// current key modifiers
19  pub mod_: Uint16,
20  /// unused padding field
21  #[allow(unused)]
22  pub unused: Uint32,
23}
24
25extern "C" {
26  /// Get the window which currently has keyboard focus.
27  pub fn SDL_GetKeyboardFocus() -> *mut SDL_Window;
28
29  /// Get a snapshot of the current state of the keyboard.
30  ///
31  /// * `numkeys` if non-NULL, receives the length of the returned array.
32  ///
33  /// **Returns:** An array of key states. Indexes into this array are obtained
34  /// by using [`SDL_Scancode`] values.
35  pub fn SDL_GetKeyboardState(numkeys: *mut c_int) -> *const u8;
36
37  /// Get the current key modifier state for the keyboard.
38  pub fn SDL_GetModState() -> SDL_Keymod;
39
40  /// Set the current key modifier state for the keyboard.
41  ///
42  /// This does not change the keyboard state, only the key modifier flags.
43  pub fn SDL_SetModState(modstate: SDL_Keymod);
44
45  /// Get the key code corresponding to the given scancode according to the
46  /// current keyboard layout.
47  ///
48  /// See [`SDL_Keycode`] for details.
49  ///
50  /// See Also: [`SDL_GetKeyName`]
51  pub fn SDL_GetKeyFromScancode(scancode: SDL_Scancode) -> SDL_Keycode;
52
53  /// Get the scancode corresponding to the given key code according to the
54  /// current keyboard layout.
55  ///
56  /// See [`SDL_Scancode`]
57  ///
58  /// See Also: [`SDL_GetScancodeName`]
59  pub fn SDL_GetScancodeFromKey(key: SDL_Keycode) -> SDL_Scancode;
60
61  /// Get a human-readable name for a scancode.
62  ///
63  /// **Returns** A pointer to the name for the scancode. If the scancode
64  /// doesn't have a name, this function returns an empty string ("").
65  pub fn SDL_GetScancodeName(scancode: SDL_Scancode) -> *const c_char;
66
67  /// Get a scancode from a human-readable name.
68  ///
69  /// **Return:** the scancode, or `SDL_SCANCODE_UNKNOWN` if the name wasn't
70  /// recognized.
71  pub fn SDL_GetScancodeFromName(name: *const c_char) -> SDL_Scancode;
72
73  /// Get a human-readable name for a key.
74  ///
75  /// **Return:** A pointer to a UTF-8 string that stays valid at least until
76  /// the next call to this function. If you need it around any longer, you must
77  /// copy it. If the key doesn't have a name, this function returns an empty
78  /// string ("").
79  pub fn SDL_GetKeyName(key: SDL_Keycode) -> *const c_char;
80
81  /// Get a key code from a human-readable name.
82  ///
83  /// **Returns:** the key code, or `SDLK_UNKNOWN` if the name wasn't
84  /// recognized.
85  pub fn SDL_GetKeyFromName(name: *const c_char) -> SDL_Keycode;
86
87  /// Start accepting Unicode text input events.
88  ///
89  /// This function will show the on-screen keyboard if supported.
90  ///
91  /// See Also: [`SDL_StopTextInput`], [`SDL_SetTextInputRect`],
92  /// [`SDL_HasScreenKeyboardSupport`]
93  pub fn SDL_StartTextInput();
94
95  /// Return whether or not Unicode text input events are enabled.
96  ///
97  /// See Also: [`SDL_StartTextInput`], [`SDL_StopTextInput`]
98  pub fn SDL_IsTextInputActive() -> SDL_bool;
99
100  /// Stop receiving any text input events.
101  ///
102  /// This function will hide the on-screen keyboard if supported.
103  ///
104  /// See Also: [`SDL_StartTextInput`], [`SDL_HasScreenKeyboardSupport`]
105  pub fn SDL_StopTextInput();
106
107  /// Set the rectangle used to type Unicode text inputs.
108  ///
109  /// This is used as a hint for IME and on-screen keyboard placement.
110  ///
111  /// See Also: [`SDL_StartTextInput`]
112  pub fn SDL_SetTextInputRect(rect: *mut SDL_Rect);
113
114  /// Returns whether the platform has some screen keyboard support.
115  ///
116  /// **Returns:** `SDL_TRUE` if some keyboard support is available, else
117  /// `SDL_FALSE`.
118  ///
119  /// **Note:** Not all screen keyboard functions are supported on all
120  /// platforms.
121  ///
122  /// See Also: [`SDL_IsScreenKeyboardShown`]
123  pub fn SDL_HasScreenKeyboardSupport() -> SDL_bool;
124
125  /// Returns whether the screen keyboard is shown for given window.
126  ///
127  /// * `window` The window for which screen keyboard should be queried.
128  ///
129  /// **Returns:** `SDL_TRUE` if screen keyboard is shown, else `SDL_FALSE`.
130  ///
131  /// See Also: [`SDL_HasScreenKeyboardSupport`]
132  pub fn SDL_IsScreenKeyboardShown(window: *mut SDL_Window) -> SDL_bool;
133}

5/// Audio format flags.
6///
7/// Use the appropriate const functions to query the bits.
8///
9/// ```txt
10/// ++-----------------------sample is signed if set
11/// ||
12/// ||       ++----------sample is big-endian if set
13/// ||       ||
14/// ||       ||          ++---sample is float if set
15/// ||       ||          ||
16/// ||       ||          || +---sample bit size---+
17/// ||       ||          || |                     |
18/// 15 14 13 12 11 10 09 08 07 06 05 04 03 02 01 00
19/// ```
20/// (Unspecified bits are always zero.)
21#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
22#[repr(transparent)]
23pub struct SDL_AudioFormat(pub u16);
24
25/// Pits per sample. eg: i16 = 16, f32 = 32.
26#[inline]
27#[must_use]
28pub const fn SDL_AUDIO_BITSIZE(af: SDL_AudioFormat) -> u16 {
29  af.0 & 0xFF
30}
31/// If the sample type is a floating type.
32#[inline]
33#[must_use]
34pub const fn SDL_AUDIO_ISFLOAT(af: SDL_AudioFormat) -> bool {
35  af.0 & (1 << 8) != 0
36}
37/// If the samples are big-endian.
38#[inline]
39#[must_use]
40pub const fn SDL_AUDIO_ISBIGENDIAN(af: SDL_AudioFormat) -> bool {
41  af.0 & (1 << 12) != 0
42}
43/// If the samples are signed values.
44#[inline]
45#[must_use]
46pub const fn SDL_AUDIO_ISSIGNED(af: SDL_AudioFormat) -> bool {
47  af.0 & (1 << 15) != 0
48}
49/// If the samples are an int type.
50#[inline]
51#[must_use]
52pub const fn SDL_AUDIO_ISINT(af: SDL_AudioFormat) -> bool {
53  !SDL_AUDIO_ISFLOAT(af)
54}
55/// If the samples are little-endian.
56#[inline]
57#[must_use]
58pub const fn SDL_AUDIO_ISLITTLEENDIAN(af: SDL_AudioFormat) -> bool {
59  !SDL_AUDIO_ISBIGENDIAN(af)
60}
61/// If the samples are unsigned values.
62#[inline]
63#[must_use]
64pub const fn SDL_AUDIO_ISUNSIGNED(af: SDL_AudioFormat) -> bool {
65  !SDL_AUDIO_ISSIGNED(af)
66}

108/// This function is called when the audio device needs more data.
109///
110/// * `userdata` An application-specific parameter saved in the `SDL_AudioSpec`
111///   structure
112/// * `stream` A pointer to the audio data buffer.
113/// * `len`    The length of that buffer in bytes.
114///
115/// The buffer is uninitialized when the callback is called.
116///
117/// The callback *must* initialize the entire buffer.
118///
119/// Once the callback returns, the buffer will no longer be valid.
120///
121/// See [`SDL_AudioSpec`] for default sample ordering.
122///
123/// You can choose to avoid callbacks and use SDL_QueueAudio() instead, if you
124/// like. Just open your audio device with a NULL callback.
125pub type SDL_AudioCallback = Option<
126  unsafe extern "C" fn(userdata: *mut c_void, stream: *mut Uint8, len: c_int),
127>;
128
129/// The calculated values in this structure are calculated by SDL_OpenAudio().
130///
131/// For multi-channel audio, the default SDL channel mapping is:
132/// ```txt
133/// 2:  FL FR                       (stereo)
134/// 3:  FL FR LFE                   (2.1 surround)
135/// 4:  FL FR BL BR                 (quad)
136/// 5:  FL FR FC BL BR              (quad + center)
137/// 6:  FL FR FC LFE SL SR          (5.1 surround - last two can also be BL BR)
138/// 7:  FL FR FC LFE BC SL SR       (6.1 surround)
139/// 8:  FL FR FC LFE BL BR SL SR    (7.1 surround)
140/// ```
141#[repr(C)]
142pub struct SDL_AudioSpec {
143  /// DSP frequency -- samples per second
144  pub freq: c_int,
145  /// Audio data format
146  pub format: SDL_AudioFormat,
147  /// Number of channels: 1 mono, 2 stereo, etc
148  pub channels: Uint8,
149  /// Audio buffer silence value (calculated)
150  pub silence: Uint8,
151  /// Audio buffer size in sample FRAMES (total samples divided by channel
152  /// count)
153  pub samples: Uint16,
154  /// Necessary for some compile environments
155  pub padding: Uint16,
156  /// Audio buffer size in bytes (calculated)
157  pub size: Uint32,
158  /// Callback that feeds the audio device (NULL to use SDL_QueueAudio()).
159  pub callback: SDL_AudioCallback,
160  /// Userdata passed to callback (ignored for NULL callbacks).
161  pub userdata: *mut c_void,
162}

168/// A structure to hold a set of audio conversion filters and buffers.
169///
170/// Note that various parts of the conversion pipeline can take advantage
171/// of SIMD operations (like SSE2, for example). SDL_AudioCVT doesn't require
172/// you to pass it aligned data, but can possibly run much faster if you
173/// set both its (buf) field to a pointer that is aligned to 16 bytes, and its
174/// (len) field to something that's a multiple of 16, if possible.
175#[repr(C)]
176pub struct SDL_AudioCVT {
177  /// Set to 1 if conversion possible
178  pub needed: c_int,
179  /// Source audio format
180  pub src_format: SDL_AudioFormat,
181  /// Target audio format
182  pub dst_format: SDL_AudioFormat,
183  /// Rate conversion increment
184  pub rate_incr: f64,
185  /// Buffer to hold entire audio data
186  pub buf: *mut Uint8,
187  /// Length of original audio buffer
188  pub len: c_int,
189  /// Length of converted audio buffer
190  pub len_cvt: c_int,
191  /// buffer must be len*len_mult big
192  pub len_mult: c_int,
193  /// Given len, final size is len*len_ratio
194  pub len_ratio: f64,
195  /// NULL-terminated list of filter functions
196  pub filters: [SDL_AudioFilter; 10],
197  /// Current audio conversion function
198  pub filter_index: c_int,
199}

254  /// This function opens the audio device with the desired parameters, and
255  /// returns 0 if successful, placing the actual hardware parameters in the
256  /// structure pointed to by `obtained`. If `obtained` is NULL, the audio
257  /// data passed to the callback function will be guaranteed to be in the
258  /// requested format, and will be automatically converted to the hardware
259  /// audio format if necessary.  This function returns -1 if it failed
260  /// to open the audio device, or couldn't set up the audio thread.
261  ///
262  /// When filling in the desired audio spec structure,
263  /// - `desired->freq` should be the desired audio frequency in samples-per-
264  ///   second.
265  /// - `desired->format` should be the desired audio format.
266  /// - `desired->samples` is the desired size of the audio buffer, in samples.
267  ///   This number should be a power of two, and may be adjusted by the audio
268  ///   driver to a value more suitable for the hardware.  Good values seem to
269  ///   range between 512 and 8096 inclusive, depending on the application and
270  ///   CPU speed.  Smaller values yield faster response time, but can lead to
271  ///   underflow if the application is doing heavy processing and cannot fill
272  ///   the audio buffer in time.  A stereo sample consists of both right and
273  ///   left channels in LR ordering. Note that the number of samples is
274  ///   directly related to time by the following formula: `ms =
275  ///   (samples*1000)/freq`
276  /// - `desired->size` is the size in bytes of the audio buffer, and is
277  ///   calculated by SDL_OpenAudio().
278  /// - `desired->silence` is the value used to set the buffer to silence, and
279  ///   is calculated by SDL_OpenAudio().
280  /// - `desired->callback` should be set to a function that will be called when
281  ///   the audio device is ready for more data.  It is passed a pointer to the
282  ///   audio buffer, and the length in bytes of the audio buffer. This function
283  ///   usually runs in a separate thread, and so you should protect data
284  ///   structures that it accesses by calling SDL_LockAudio() and
285  ///   SDL_UnlockAudio() in your code. Alternately, you may pass a NULL pointer
286  ///   here, and call SDL_QueueAudio() with some frequency, to queue more audio
287  ///   samples to be played (or for capture devices, call SDL_DequeueAudio()
288  ///   with some frequency, to obtain audio samples).
289  /// - `desired->userdata` is passed as the first parameter to your callback
290  ///   function. If you passed a NULL callback, this value is ignored.
291  ///
292  /// The audio device starts out playing silence when it's opened, and should
293  /// be enabled for playing by calling [`SDL_PauseAudio(0)`](SDL_PauseAudio)
294  /// when you are ready  for your audio callback function to be called.
295  /// Since the audio driver  may modify the requested size of the audio
296  /// buffer, you should allocate  any local mixing buffers after you open the
297  /// audio device.
298  pub fn SDL_OpenAudio(
299    desired: *mut SDL_AudioSpec, obtained: *mut SDL_AudioSpec,
300  ) -> c_int;

372  /// Load the audio data of a WAVE file into memory
373  ///
374  /// Loading a WAVE file requires `src`, `spec`, `audio_buf`, and `audio_len`
375  /// to be valid pointers. The entire data portion of the file is then loaded
376  /// into memory and decoded if necessary.
377  ///
378  /// If `freesrc` is non-zero, the data source gets automatically closed and
379  /// freed before the function returns.
380  ///
381  /// Supported are RIFF WAVE files with the formats PCM (8, 16, 24, and 32
382  /// bits), IEEE Float (32 bits), Microsoft ADPCM and IMA ADPCM (4 bits), and
383  /// A-law and µ-law (8 bits). Other formats are currently unsupported and
384  /// cause an error.
385  ///
386  /// If this function succeeds, the pointer returned by it is equal to `spec`
387  /// and the pointer to the audio data allocated by the function is written to
388  /// `audio_buf` and its length in bytes to `audio_len`. The [`SDL_AudioSpec`]
389  /// members `freq`, `channels`, and `format` are set to the values of the
390  /// audio data in the buffer. The `samples` member is set to a sane default
391  /// and all others are set to zero.
392  ///
393  /// It's necessary to use [`SDL_FreeWAV`] to free the audio data returned in
394  /// `audio_buf` when it is no longer used.
395  ///
396  /// Because of the under-specification of the Waveform format, there are many
397  /// problematic files in the wild that cause issues with strict decoders. To
398  /// provide compatibility with these files, this decoder is lenient in regards
399  /// to the truncation of the file, the fact chunk, and the size of the RIFF
400  /// chunk. The hints SDL_HINT_WAVE_RIFF_CHUNK_SIZE, SDL_HINT_WAVE_TRUNCATION,
401  /// and SDL_HINT_WAVE_FACT_CHUNK can be used to tune the behavior of the
402  /// loading process.
403  ///
404  /// Any file that is invalid (due to truncation, corruption, or wrong values
405  /// in the headers), too big, or unsupported causes an error. Additionally,
406  /// any critical I/O error from the data source will terminate the loading
407  /// process with an error. The function returns NULL on error and in all
408  /// cases (with the exception of `src` being NULL), an appropriate error
409  /// message will be set.
410  ///
411  /// It is required that the data source supports seeking.
412  ///
413  /// Example:
414  /// ```txt
415  /// SDL_LoadWAV_RW(SDL_RWFromFile("sample.wav", "rb"), 1, ...);
416  /// ```
417  ///
418  /// * `src` The data source with the WAVE data
419  /// * `freesrc` A integer value that makes the function close the data source
420  ///   if non-zero
421  /// * `spec` A pointer filled with the audio format of the audio data
422  /// * `audio_buf` A pointer filled with the audio data allocated by the
423  ///   function
424  /// * `audio_len` A pointer filled with the length of the audio data buffer in
425  ///   bytes
426  ///
427  /// **Return:** NULL on error, or non-NULL on success.
428  pub fn SDL_LoadWAV_RW(
429    src: *mut SDL_RWops, freesrc: c_int, spec: *mut SDL_AudioSpec,
430    audio_buf: *mut *mut Uint8, audio_len: *mut Uint32,
431  ) -> *mut SDL_AudioSpec;

549  /// Queue more audio on non-callback devices.
550  ///
551  /// (If you are looking to retrieve queued audio from a non-callback capture
552  /// device, you want [`SDL_DequeueAudio`] instead. This will return -1 to
553  /// signify an error if you use it with capture devices.)
554  ///
555  /// SDL offers two ways to feed audio to the device: you can either supply a
556  /// callback that SDL triggers with some frequency to obtain more audio
557  /// (pull method), or you can supply no callback, and then SDL will expect
558  /// you to supply data at regular intervals (push method) with this function.
559  ///
560  /// There are no limits on the amount of data you can queue, short of
561  /// exhaustion of address space. Queued data will drain to the device as
562  /// necessary without further intervention from you. If the device needs
563  /// audio but there is not enough queued, it will play silence to make up
564  /// the difference. This means you will have skips in your audio playback
565  /// if you aren't routinely queueing sufficient data.
566  ///
567  /// This function copies the supplied data, so you are safe to free it when
568  /// the function returns. This function is thread-safe, but queueing to the
569  /// same device from two threads at once does not promise which buffer will
570  /// be queued first.
571  ///
572  /// You may not queue audio on a device that is using an application-supplied
573  /// callback; doing so returns an error. You have to use the audio callback
574  /// or queue audio with this function, but not both.
575  ///
576  /// You should **not** call [`SDL_LockAudio`] on the device before queueing;
577  /// SDL handles locking internally for this function.
578  ///
579  /// * `dev` The device ID to which we will queue audio.
580  /// * `data` The data to queue to the device for later playback.
581  /// * `len` The number of bytes (not samples!) to which (data) points.
582  ///
583  /// **Return:** 0 on success, or -1 on error.
584  pub fn SDL_QueueAudio(
585    dev: SDL_AudioDeviceID, data: *const c_void, len: Uint32,
586  ) -> c_int;

588  /// Dequeue more audio on non-callback devices.
589  ///
590  /// (If you are looking to queue audio for output on a non-callback playback
591  /// device, you want [``SDL_QueueAudio`] instead. This will always return 0
592  /// if you use it with playback devices.)
593  ///
594  /// SDL offers two ways to retrieve audio from a capture device: you can
595  /// either supply a callback that SDL triggers with some frequency as the
596  /// device records more audio data, (push method), or you can supply no
597  /// callback, and then SDL will expect you to retrieve data at regular
598  /// intervals (pull method) with this function.
599  ///
600  /// There are no limits on the amount of data you can queue, short of
601  /// exhaustion of address space. Data from the device will keep queuing as
602  /// necessary without further intervention from you. This means you will
603  /// eventually run out of memory if you aren't routinely dequeueing data.
604  ///
605  /// Capture devices will not queue data when paused; if you are expecting
606  /// to not need captured audio for some length of time, use
607  /// SDL_PauseAudioDevice() to stop the capture device from queueing more
608  /// data. This can be useful during, say, level loading times. When
609  /// un-paused, capture devices will start queueing data from that point,
610  /// having flushed any capturable data available while paused.
611  ///
612  /// This function is thread-safe, but dequeueing from the same device from
613  /// two threads at once does not promise which thread will dequeued data
614  /// first.
615  ///
616  /// You may not dequeue audio from a device that is using an
617  /// application-supplied callback; doing so returns an error. You have to use
618  /// the audio callback, or dequeue audio with this function, but not both.
619  ///
620  /// You should not call SDL_LockAudio() on the device before queueing; SDL
621  /// handles locking internally for this function.
622  ///
623  /// * `dev` The device ID from which we will dequeue audio.
624  /// * `data` A pointer into where audio data should be copied.
625  /// * `len` The number of bytes (not samples!) to which (data) points.
626  ///
627  /// **Return:** number of bytes dequeued, which could be less than requested.
628  pub fn SDL_DequeueAudio(
629    dev: SDL_AudioDeviceID, data: *mut c_void, len: Uint32,
630  ) -> Uint32;

632  /// Get the number of bytes of still-queued audio.
633  ///
634  /// * For playback device:
635  ///  * This is the number of bytes that have been queued for playback with
636  ///    SDL_QueueAudio(), but have not yet been sent to the hardware. This
637  ///    number may shrink at any time, so this only informs of pending data.
638  ///  * Once we've sent it to the hardware, this function can not decide the
639  ///    exact byte boundary of what has been played. It's possible that we just
640  ///    gave the hardware several kilobytes right before you called this
641  ///    function, but it hasn't played any of it yet, or maybe half of it, etc.
642  ///
643  /// * For capture devices:
644  ///   * This is the number of bytes that have been captured by the device and
645  ///     are waiting for you to dequeue. This number may grow at any time, so
646  ///     this only informs of the lower-bound of available data.
647  ///
648  /// You may not queue audio on a device that is using an application-supplied
649  /// callback; calling this function on such a device always returns 0. You
650  /// have to queue audio with SDL_QueueAudio()/SDL_DequeueAudio(), or use the
651  /// audio callback, but not both.
652  ///
653  /// You should not call SDL_LockAudio() on the device before querying; SDL
654  /// handles locking internally for this function.
655  ///
656  /// * `dev` The device ID of which we will query queued audio size.
657  ///
658  /// **Return:** Number of bytes (not samples!) of queued audio.
659  pub fn SDL_GetQueuedAudioSize(dev: SDL_AudioDeviceID) -> Uint32;

661  /// Drop any queued audio data.
662  ///
663  /// For playback devices, this is any queued data
664  /// still waiting to be submitted to the hardware. For capture devices, this
665  /// is any data that was queued by the device that hasn't yet been dequeued by
666  /// the application.
667  ///
668  /// Immediately after this call, [`SDL_GetQueuedAudioSize`] will return 0. For
669  /// playback devices, the hardware will start playing silence if more audio
670  /// isn't queued. Un-paused capture devices will start filling the queue again
671  /// as soon as they have more data available (which, depending on the state
672  /// of the hardware and the thread, could be before this function call
673  /// returns!).
674  ///
675  /// This will not prevent playback of queued audio that's already been sent
676  /// to the hardware, as we can not undo that, so expect there to be some
677  /// fraction of a second of audio that might still be heard. This can be
678  /// useful if you want to, say, drop any pending music during a level change
679  /// in your game.
680  ///
681  /// You may not queue audio on a device that is using an application-supplied
682  /// callback; calling this function on such a device is always a no-op.
683  /// You have to queue audio with [`SDL_QueueAudio`]/[`SDL_DequeueAudio`], or
684  /// use the audio callback, but not both.
685  ///
686  /// You should not call SDL_LockAudio() on the device before clearing the
687  /// queue; SDL handles locking internally for this function.
688  ///
689  /// This function always succeeds and thus returns void.
690  ///
691  /// * `dev` The device ID of which to clear the audio queue.
692  pub fn SDL_ClearQueuedAudio(dev: SDL_AudioDeviceID);

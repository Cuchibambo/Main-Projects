1//! SDL's portable 2D rendering functions.
2//!
3//! This API supports the following features:
4//! * single pixel points
5//! * single pixel lines
6//! * filled rectangles
7//! * texture images
8//!
9//! The primitives may be drawn in opaque, blended, or additive modes.
10//!
11//! The texture images may be drawn in opaque, blended, or additive modes. They
12//! can have an additional color tint or alpha modulation applied to them, and
13//! may also be stretched with linear interpolation.
14//!
15//! This API is designed to accelerate simple 2D operations. You may want more
16//! functionality such as polygons and particle effects and in that case you
17//! should use SDL's OpenGL/Direct3D support or one of the many good 3D engines.
18//!
19//! These functions must be called from the main thread. See this bug for
20//! details: <http://bugzilla.libsdl.org/show_bug.cgi?id=1995>
21

30/// Flags used when creating a rendering context
31#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
32#[repr(transparent)]
33pub struct SDL_RendererFlags(pub u32);
34impl_bit_ops_for_tuple_newtype!(SDL_RendererFlags);
35/// The renderer is a software fallback
36pub const SDL_RENDERER_SOFTWARE: SDL_RendererFlags =
37  SDL_RendererFlags(0x00000001);
38/// The renderer uses hardware acceleration
39pub const SDL_RENDERER_ACCELERATED: SDL_RendererFlags =
40  SDL_RendererFlags(0x00000002);
41/// Present is synchronized with the refresh rate
42pub const SDL_RENDERER_PRESENTVSYNC: SDL_RendererFlags =
43  SDL_RendererFlags(0x00000004);
44/// The renderer supports rendering to texture
45pub const SDL_RENDERER_TARGETTEXTURE: SDL_RendererFlags =
46  SDL_RendererFlags(0x00000008);

48/// Information on the capabilities of a render driver or context.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
50#[repr(C)]
51#[allow(missing_docs)]
52pub struct SDL_RendererInfo {
53  /// The name of the renderer (borrowed)
54  pub name: *const c_char,
55  /// Supported [`SDL_RendererFlags`]
56  pub flags: Uint32,
57  /// The number of available texture formats
58  pub num_texture_formats: Uint32,
59  /// The available texture formats
60  pub texture_formats: [Uint32; 16],
61  /// The maximum texture width
62  pub max_texture_width: c_int,
63  /// The maximum texture height
64  pub max_texture_height: c_int,
65}
66impl Default for SDL_RendererInfo {
67  #[inline]
68  #[must_use]
69  fn default() -> Self {
70    unsafe { core::mem::zeroed() }
71  }
72}

996  /// This is for a very-specific case: if you are using SDL's render API,

1//! Quit stuff.
2//!
3//! An [`SDL_QUIT`] event is generated when the user tries to close the
4//! application window. If it is ignored or filtered out, the window will remain
5//! open. If it is not ignored or filtered, it is queued normally and the window
6//! is allowed to close.  When the window is closed, screen updates will
7//! complete, but have no effect.
8//!
9//! [`SDL_Init`] installs signal handlers for `SIGINT` (keyboard interrupt) and
10//! `SIGTERM` (system termination request), if handlers do not already exist,
11//! that generate [`SDL_QUIT`] events as well.  There is no way to determine the
12//! cause of an [`SDL_QUIT`] event, but setting a signal handler in your
13//! application will override the default generation of quit events for that
14//! signal.
15
16use crate::events::*;
17
18// makes rustdoc link properly!
19#[allow(unused)]
20use crate::*;
21
22/// Checks if a quit event is currently in the queue.
23#[inline]
24#[must_use]
25pub unsafe fn SDL_QuitRequested() -> bool {
26  SDL_PumpEvents();
27  SDL_PeepEvents(core::ptr::null_mut(), 0, SDL_PEEKEVENT, SDL_QUIT, SDL_QUIT)
28    > 0
29}

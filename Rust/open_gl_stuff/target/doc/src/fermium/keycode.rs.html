11/// The SDL keyboard keycode representation.
12///
13/// This is used in places like the [`SDL_Keysym`] struct for the "software" /
14/// "logical" designation of a key.
15///
16/// Values of this type are used to represent keyboard keys using the current
17/// layout of the keyboard.  These values include Unicode values representing
18/// the unmodified character that would be generated by pressing the key, or an
19/// SDLK_* constant for those keys that do not generate characters.
20///
21/// A special exception is the number keys at the top of the keyboard which
22/// always map to `SDLK_0` through `SDLK_9`, regardless of layout.
23///
24/// See all the constants named `SDLK_*`
25#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
26#[repr(transparent)]
27pub struct SDL_Keycode(pub i32);
28
29pub const SDLK_SCANCODE_MASK: i32 = 1 << 30;
30
31/// Converts a scancode into a keycode.
32///
33/// Primarily intended for internal use.
34#[inline]
35#[must_use]
36pub const fn SDL_SCANCODE_TO_KEYCODE(x: SDL_Scancode) -> SDL_Keycode {
37  SDL_Keycode(x.0 | (1 << 30))
38}

372/// Representation of modifier key flags.
373///
374/// This is used in places like the [`SDL_Keysym`] struct.
375///
376/// See all the constants named `KMOD_*`
377#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
378#[repr(transparent)]
379pub struct SDL_Keymod(pub i32);
380impl_bit_ops_for_tuple_newtype!(SDL_Keymod);
381impl SDL_Keymod {
382  /// Convenience function to downcast into `u16`.
383  ///
384  /// This is needed for easy interaction with the `mod_` field of
385  /// [`SDL_Keysym`].
386  #[inline]
387  #[must_use]
388  pub const fn as_u16(self) -> u16 {
389    self.0 as i16 as u16
390  }
391}
